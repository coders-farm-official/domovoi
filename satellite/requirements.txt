# Pi satellite client deps. Install inside the venv created in
# satellite/PROVISIONING.md step 6:
#
#   source ~/satellite-venv/bin/activate
#   pip install -r satellite/requirements.txt
#   pip install --no-deps "openwakeword>=0.6.0"
#
# numpy/sounddevice/webrtcvad-wheels carry C extensions; ARM wheels exist for
# all of them on PyPI, so a pip install on Pi Zero 2 W under Bookworm/Trixie
# should not require a compiler.
#
# openwakeword 0.6.0 pins tflite-runtime as a hard Linux dep, but no
# tflite-runtime wheel exists for Python 3.13 (Trixie's default). client.py
# uses the ONNX inference path, so we install openwakeword's runtime deps
# explicitly here and install openwakeword itself with --no-deps (see above).

numpy>=1.26
sounddevice>=0.4.6
webrtcvad-wheels>=2.0.14
websockets>=12.0

# State-indicator LEDs on the ReSpeaker HAT (APA102 over SPI). Tiny C
# extension; ships aarch64 wheels for cp313. If you're running on
# non-ReSpeaker hardware, leave the package installed but set
# `[leds] enabled = false` in ~/.domovoi/config.toml.
spidev>=3.6

# openwakeword runtime deps (installed explicitly; openwakeword itself goes
# in with --no-deps to skip its tflite-runtime pin).
#
# onnxruntime is pinned to 1.20.0: cp313 wheels start at 1.20, and 1.21+ abort
# on Pi Zero 2 W's Cortex-A53 with a vector::operator[] assertion from the
# manylinux gcc-toolset-14 build. 1.20.0 is the only known-good cp313+aarch64
# combo for this hardware.
onnxruntime==1.20.0
tqdm>=4.0,<5.0
scipy>=1.3,<2
scikit-learn>=1,<2
requests>=2.0,<3

# Ed25519, for checking that the server answering us is the one this card
# was prepared for (satellite/server_identity.py) and that a code or
# plugin-payload manifest carries that server's signature.
#
# 64-bit Pi OS (aarch64, the supported build) gets a cp39-abi3
# manylinux_2_28_aarch64 wheel — no Rust toolchain, no compiler. A 32-bit
# armv7l image has NO wheel and would try to build Rust, so this install is
# allowed to fail there: satellite/_ed25519.py is a vendored pure-Python
# implementation of the same thing and the code falls back to it
# automatically. Install with `|| true` in a 32-bit bootstrap.
cryptography>=42.0
